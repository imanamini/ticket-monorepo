const convertEnglishDigitsToPersian = (string) => {
  // make sure it is string
  string = '' + string;
  let map = {
    '0': '۰',
    '1': '۱',
    '2': '۲',
    '3': '۳',
    '4': '۴',
    '5': '۵',
    '6': '۶',
    '7': '۷',
    '8': '۸',
    '9': '۹',
  };

  let numbers = string.split('');
  for (let i = 0; i < numbers.length; i++) {
    if (map.hasOwnProperty(numbers[i])) {
      numbers[i] = map[numbers[i]];
    }
  }

  return numbers.join('');
};
